⚪ กำลังตรวจสอบ

เชื่อมต่อชีตเดียว บันทึกทั้ง คะแนนฝึกซ้อม (แผ่น 1) และ ข้อมูลสุขภาพ BMI (แผ่น 2) อัตโนมัติ

📝 โค้ด Google Apps Script รองรับ 2 แผ่นงานในชีตเดิม (Auto-Provisioning):

ใน Google Sheets เดิมของคุณครู ไปที่เมนู ส่วนขยาย (Extensions) ➔ Apps Script ลบโค้ดเดิมทั้งหมดแล้ววางโค้ดชุดนี้: 

function doGet(e) {
  return ContentService.createTextOutput("✅ Webhook PECLASS ทำงานปกติ พร้อมรับข้อมูล!");
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var d = JSON.parse(e.postData.contents);

    // 1. ถ้าเป็นข้อมูลสุขภาพ & BMI ให้บันทึกลงแผ่นงานที่ 2
    if (d.type === "health_bmi") {
      var healthSheet = ss.getSheetByName("ข้อมูลสุขภาพ_BMI");
      if (!healthSheet) {
        healthSheet = ss.insertSheet("ข้อมูลสุขภาพ_BMI");
        healthSheet.appendRow(["วัน-เวลา", "ห้อง", "เลขที่", "ชื่อ-นามสกุล", "เพศ", "อายุ (ปี)", "น้ำหนัก (กก.)", "ส่วนสูง (ซม.)", "ค่า BMI", "เกณฑ์การเจริญเติบโต"]);
        healthSheet.getRange("A1:J1").setBackground("#10b981").setFontColor("#ffffff").setFontWeight("bold");
      }
      healthSheet.appendRow([d.timestamp, d.room, d.studentNumber, d.studentName, d.gender, d.age, d.weight, d.height, d.bmi, d.classification]);
      return ContentService.createTextOutput("OK_HEALTH");
    }

    // 2. ถ้าเป็นคะแนนการฝึกซ้อม ให้บันทึกลงแผ่นงานแรกตามปกติ
    var scoreSheet = ss.getSheetByName("คะแนนการฝึกซ้อม") || ss.getSheets()[0];
    if (scoreSheet.getLastRow() === 0) {
      scoreSheet.appendRow(["วัน-เวลา", "ห้อง", "เลขที่", "ชื่อ-นามสกุล", "รหัสท่า", "จำนวนครั้ง", "เป้าหมาย", "ความเป๊ะ (%)", "เกรด", "XP ที่ได้"]);
      scoreSheet.getRange("A1:J1").setBackground("#6366f1").setFontColor("#ffffff").setFontWeight("bold");
    }
    scoreSheet.appendRow([d.timestamp, d.room, d.studentNumber, d.studentName, d.exerciseId, d.reps, d.targetReps, d.accuracy, d.grade, d.xpEarned]);
    return ContentService.createTextOutput("OK_SCORE");
  } catch (err) {
    return ContentService.createTextOutput("ERROR: " + err.toString());
  }
}
⚠️ สิ่งสำคัญที่สุดในการ Deploy:
1. กด Deploy (การทำให้ใช้งานได้) ➔ Manage deployments (จัดการการทำให้ใช้งานได้)
2. กดไอคอนดินสอ ➔ ตรง Version เลือก New version (เวอร์ชันใหม่) ทุกครั้งที่มีการแก้โค้ด
3. ตรง Who has access (ผู้มีสิทธิ์เข้าถึง) ต้องเลือกเป็น Anyone (ทุกคน) เสมอ!
4. นำ URL ที่ลงท้ายด้วย /exec มาวางในช่องด้านบนแล้วกด "บันทึก Webhook"

ซิงค์ตารางคะแนนสด Realtime ข้ามอุปกรณ์ และบันทึกรูปถ่ายหลักฐานการทำท่าทางของเด็ก

🛠️ ขั้นตอนเชื่อมต่อ Supabase:

  1. เข้าเว็บ supabase.com ล็อกอินและเปิดโปรเจกต์ของคุณ
  2. ไปที่เมนู Project Settings ➔ API นำ Project URL และ anon key มาวาง
  3. เปิดไฟล์ supabase_schema.sql คัดลอกโค้ดไปวางใน SQL Editor แล้วกด Run
💡 ซิงค์คะแนน & สุขภาพสด (Realtime): ไฟล์ supabase_schema.sql ได้รวมตาราง health_records ไว้แล้ว หากเคยรันโค้ดเก่า ให้นำส่วนของตารางนี้ไปกด Run เพิ่มใน Supabase เพื่อให้ข้อมูลสุขภาพที่เด็กบันทึกจากมือถือไหลเข้า Dashboard ทันที
